        function getProduct(id) { return PRODUCTS.find(p => p.id === id); }

        function getProjectByKey(key) {
            if (key === 'bcg') return PROJECTS_DATA.find(p => p.name.includes('BCG'));
            if (key === 'bnp') return PROJECTS_DATA.find(p => p.name.includes('Banque'));
            return null;
        }

        function getProjectInventory(projectKey) {
            const project = PROJECTS_DATA.find(p => {
                const key = p.name.toLowerCase().replace(/\s/g, '');
                return key.includes(projectKey) || projectKey.includes(key);
            });
            if (!project) return [];
            return project.items.map(id => {
                const prod = getProduct(id);
                if (!prod) return null;
                const stockPct = prod.inStock / prod.stock;
                let status = 'OK',
                    statusClass = 'green';
                if (stockPct < 0.3) { status = 'Critical';
                    statusClass = 'red'; } else if (stockPct < 0.6) { status = 'Low';
                    statusClass = 'orange'; }
                return { ...prod, qty: `${prod.inStock}/${prod.stock}`, status, statusClass };
            }).filter(Boolean);
        }

        function getStatusCounts() {
            const critical = [];
            const low = [];
            PRODUCTS.forEach(p => {
                const pct = p.inStock / p.stock;
                if (pct < 0.3) critical.push(p.name);
                else if (pct < 0.6) low.push(p.name);
            });
            return { critical, low, total: critical.length + low.length };
        }

        function generateId() { return Date.now().toString(36) + Math.random().toString(36).slice(2, 6); }

        function getTimestamp() { return new Date().toISOString(); }

        function formatDate(d) {
            if (!d) return 'N/A';
            const dt = new Date(d);
            return dt.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
        }

        function timeAgo(date) {
            const diff = Date.now() - new Date(date).getTime();
            const mins = Math.floor(diff / 60000);
            if (mins < 1) return 'Just now';
            if (mins < 60) return `${mins}m ago`;
            const hrs = Math.floor(mins / 60);
            if (hrs < 24) return `${hrs}h ago`;
            return `${Math.floor(hrs / 24)}d ago`;
        }

                if (label) label.textContent = categories[idx] || 'Other';

        function addToProposal(productId) {
            if (!currentProposal) {
                currentProposal = {
                    id: 'prop_' + generateId(),
                    client: document.getElementById('proposalClient').value || 'New Client',
                    project: document.getElementById('proposalProject').value || 'New Project',
                    date: document.getElementById('proposalDate').value || getTimestamp().slice(0, 10),
                    items: [],
                    status: 'draft',
                    createdAt: getTimestamp()
                };
            }
            const existing = currentProposal.items.find(i => i.productId === productId);
            if (existing) { existing.qty += 1; } else { currentProposal.items.push({ productId, qty: 1 }); }
            renderProposalCanvas();
            document.getElementById('proposalBuilderContainer').style.display = 'block';
            document.getElementById('builderTitle').innerText = editingProposalId ? '✏️ Edit Proposal' : '✏️ New Proposal';
            showToast(`✅ Added ${getProduct(productId)?.name || 'item'} to proposal`, 'success', '➕');
            addActivity('Added item to proposal', `${getProduct(productId)?.name || 'Item'}`, '➕');
        }

        function removeFromProposal(productId) {
            if (!currentProposal) return;
            currentProposal.items = currentProposal.items.filter(i => i.productId !== productId);
            renderProposalCanvas();
        }

        function updateProposalQty(productId, delta) {
            if (!currentProposal) return;
            const item = currentProposal.items.find(i => i.productId === productId);
            if (!item) return;
            item.qty = Math.max(1, item.qty + delta);
            renderProposalCanvas();
        }

        function renderProposalCanvas() {
            const container = document.getElementById('canvasItems');
            if (!container) return;
            if (!currentProposal || currentProposal.items.length === 0) {
                container.innerHTML = `
                        <div class="empty-state">
                            <div class="empty-icon">🛒</div>
                            <h4>Your proposal is empty</h4>
                            <p>Browse the catalog and click <strong>+ Add</strong> to build your proposal.</p>
                        </div>
                    `;
                document.getElementById('canvasSubtotal').innerText = formatPrice(0);
                document.getElementById('canvasMarkup').innerText = formatPrice(0);
                document.getElementById('canvasGrandTotal').innerText = formatPrice(0);
                return;
            }

            let subtotal = 0;
            container.innerHTML = currentProposal.items.map(item => {
                const prod = getProduct(item.productId);
                if (!prod) return '';
                const lineTotal = prod.price * item.qty;
                subtotal += lineTotal;
                return `
                        <div class="canvas-item">
                            <div class="ci-thumb">${prod.emoji}</div>
                            <div class="ci-info">
                                <div class="ci-name">${prod.name}</div>
                                <div class="ci-supplier">${prod.supplier} · ${formatPrice(prod.price)}/u</div>
                            </div>
                            <div class="ci-qty">
                                <button onclick="updateProposalQty('${prod.id}', -1)">−</button>
                                <span>${item.qty}</span>
                                <button onclick="updateProposalQty('${prod.id}', 1)">+</button>
                            </div>
                            <div class="ci-price">${formatPrice(lineTotal)}</div>
                            <button class="ci-remove" onclick="removeFromProposal('${prod.id}')">✕</button>
                        </div>
                    `;
            }).join('');

            const markup = subtotal * 0.3;
            const grand = subtotal + markup;
            document.getElementById('canvasSubtotal').innerText = formatPrice(subtotal);
            document.getElementById('canvasMarkup').innerText = formatPrice(markup);
            document.getElementById('canvasGrandTotal').innerText = formatPrice(grand);
        }

        async function saveCurrentProposal() {
            if (!currentProposal || currentProposal.items.length === 0) {
                showToast('Please add at least one item to the proposal.', 'warning', '⚠️');
                return;
            }
            currentProposal.client = document.getElementById('proposalClient').value || 'New Client';
            currentProposal.project = document.getElementById('proposalProject').value || 'New Project';
            currentProposal.date = document.getElementById('proposalDate').value || getTimestamp().slice(0, 10);
            currentProposal.status = 'draft';
            if (!currentProposal.createdAt) currentProposal.createdAt = getTimestamp();

            const idx = proposals.findIndex(p => p.id === currentProposal.id);
            if (idx !== -1) { proposals[idx] = { ...currentProposal }; } else { proposals.push({ ...currentProposal }); }
            try { localStorage.setItem('future_os_proposals', JSON.stringify(proposals)); } catch (e) {}
            showToast(`✅ Proposal saved for ${currentProposal.client}`, 'success', '💾');
            addActivity('Saved proposal', `${currentProposal.client} · ${currentProposal.project}`, '💾');
            renderProposalList();
            closeProposalBuilder();
        }

        function renderCashFlow() {
            const now = new Date();
            const thirtyDaysAgo = new Date(now);
            thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);

            const cashIn = invoices
                .filter(inv => new Date(inv.date) >= thirtyDaysAgo && inv.status === 'Paid')
                .reduce((s, inv) => s + inv.amount, 0);

            const cashOut = expenses
                .filter(e => new Date(e.date) >= thirtyDaysAgo && e.status === 'Paid')
                .reduce((s, e) => s + e.amount, 0);

            document.getElementById('cashInflow').innerText = formatPrice(cashIn);
            document.getElementById('cashOutflow').innerText = formatPrice(cashOut);
            const net = cashIn - cashOut;
            document.getElementById('netCashFlow').innerText = formatPrice(net);
            document.getElementById('netCashFlow').style.color = net >= 0 ? '#1e7a1e' : '#d32f2f';
        }

        function renderBudget() {
            const container = document.getElementById('budgetList');
            if (!container) return;

            if (entries.length === 0) {
                container.innerHTML =
                    `<div class="empty-state"><div class="empty-icon">📊</div><h4>No budget data</h4><p>Set up budget categories to track performance.</p></div>`;
                return;
            }
            container.innerHTML = entries.map(([category, data]) => {
                const variance = data.actual - data.budget;
                const pct = data.budget > 0 ? (data.actual / data.budget) * 100 : 0;
                const isOver = variance > 0;
                return `
                        <div class="budget-bar">
                            <div class="bb-label">${category}</div>

        function exportFinanceData() {
            let csv = 'Type,Title/Client,Amount,Date,Status\n';
            expenses.forEach(e => {
                csv += `Expense,${e.title},${e.amount},${e.date},${e.status}\n`;
            });
            invoices.forEach(inv => {
                csv += `Invoice,${inv.client},${inv.amount},${inv.date},${inv.status}\n`;
            });
            bills.forEach(b => {
                csv += `Bill,${b.supplier},${b.amount},${b.date},${b.status}\n`;
            });
            const blob = new Blob([csv], { type: 'text/csv' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `finance_data_${getTimestamp().slice(0,10)}.csv`;
            a.click();
            URL.revokeObjectURL(url);
            showToast('📥 Financial data exported as CSV', 'success', '📥');
            addActivity('Exported financial data', 'CSV download', '📥');
        }

        (async function init() {
            try {
                const savedProposals = localStorage.getItem('future_os_proposals');
                if (savedProposals) proposals = JSON.parse(savedProposals);
                const savedActivity = localStorage.getItem('future_os_activity');
                if (savedActivity) activityLog = JSON.parse(savedActivity);
                const savedAudit = localStorage.getItem('future_os_audit');
                if (savedAudit) auditTrail = JSON.parse(savedAudit);
                const savedDark = localStorage.getItem('future_os_darkmode');
                if (savedDark) {
                    isDarkMode = JSON.parse(savedDark);
                    if (isDarkMode) {
                        document.body.classList.add('dark-mode');
                        document.getElementById('themeIcon').innerText = '☀️';
                    }
                }
            } catch (e) {}

            document.getElementById('proposalDate').value = getTimestamp().slice(0, 10);
            document.getElementById('invoiceDate').value = getTimestamp().slice(0, 10);
            document.getElementById('billDate').value = getTimestamp().slice(0, 10);

            renderInventory();
            renderLogistics();
            renderProposalList();
            renderProjectPnL();
            renderFullInventory();
            renderProcurement();
            renderAfterSales();
            renderCatalog('catalogGrid', 'catalogSearch', 'catalogFilters', false);
            renderCatalog('catalogFullGrid', 'catalogFullSearch', 'catalogFullFilters', true);
            renderActivityLog();
            renderFinance();
            renderTeam();
            updateCurrencyUI();
            updateClock();
            updateOnlineUsers();
            updateWidgets();
            setupFinanceForms();
            setupTeamForm();

            setInterval(updateClock, 30000);
            setInterval(updateOnlineUsers, 45000);
            setInterval(() => { document.getElementById('lastSync').innerText = 'just now'; }, 60000);
            setInterval(() => {
                // Auto-advance recurring expenses
                const today = new Date();
                recurringExpenses.forEach(r => {
                    const next = new Date(r.nextDate);
                    if (next <= today) {
                        // Create an expense from recurring
                        expenses.push({
                            id: 'e' + generateId(),
                            title: r.title + ' (recurring)',
                            amount: r.amount,
                            category: 'Recurring',
                            date: getTimestamp().slice(0, 10),
                            status: 'Pending',
                            approval: 'Pending',
                            createdBy: 'System (Auto)'
                        });
                        // Advance next date
                        if (r.frequency === 'Monthly') next.setMonth(next.getMonth() + 1);
                        else if (r.frequency === 'Quarterly') next.setMonth(next.getMonth() + 3);
                        else if (r.frequency === 'Yearly') next.setFullYear(next.getFullYear() + 1);
                        r.nextDate = next.toISOString().slice(0, 10);
                        renderFinance();
                        addActivity('Auto-created recurring expense', `${r.title} · ${formatPrice(r.amount)}`,
                        '🔄');
                    }
                });
            }, 60000);

            if (activityLog.length === 0) {
                addActivity('System started', 'Vision & Cost v4.0 Enterprise loaded', '🚀');
            }

            showToast('🚀 Future OS v4.0 Enterprise · Full financial suite loaded', 'success', '🚀');
            console.log('🔥 Future OS · Vision & Cost v4.0 Enterprise loaded.');
            console.log(`📄 ${proposals.length} proposals, ${PRODUCTS.length} products, ${PROJECTS_DATA.length} projects`);
            console.log(`💰 ${expenses.length} expenses, ${invoices.length} invoices, ${bills.length} bills, ${teamMembers.length} team members`);
            console.log(`🔄 ${recurringExpenses.length} recurring expenses, ${Object.keys(budgetData).length} budget categories`);
            console.log(`📋 ${auditTrail.length} audit trail entries`);
        })();
